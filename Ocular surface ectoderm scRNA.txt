E8P5_10P5 <- readRDS("EyeRegion_E8P5_10P5.rds")
Count1 <- E6P5_E7P5_cells@assays$RNA$counts
anno1 <- E6P5_E7P5_cells@meta.data
Count2 <- E8P5_10P5@assays$RNA$counts
anno2 <- E8P5_10P5@meta.data
Seu1 <- CreateSeuratObject(counts = Count1,meta.data = anno1,project = "E6P5_E7P5", min.cells = 3, min.features = 200)
Seu2 <- CreateSeuratObject(counts = Count2,meta.data = anno2,project = "E8P5_10P5", min.cells = 3, min.features = 200)
seurat_combined <- merge(Seu1, y = Seu2)
seurat_combined <- NormalizeData(seurat_combined, normalization.method = "LogNormalize", scale.factor = 1e4)
seurat_combined <- FindVariableFeatures(seurat_combined, selection.method = 'vst', nfeatures = 2000)
seurat_combined <- ScaleData(seurat_combined)
seurat_combined <- RunPCA(seurat_combined, features = VariableFeatures(object = seurat_combined))
seurat_combined <- FindNeighbors(seurat_combined, dims = 1:15)
seurat_combined<- FindClusters(seurat_combined, resolution = 0.6)
seurat_combined <- RunUMAP(seurat_combined, dims = 1:10)
DimPlot(seurat_combined, reduction = 'umap', label = T)
DimPlot(seurat_combined, reduction = 'umap', group.by  = "stage" , label = T)
seurat_combined <- JoinLayers(seurat_combined)
combined2 <- seurat_combined %>% RunHarmony("stage", plot_convergence = T)
ose2 <- SetIdent(combined2,value = "stage")

### Pseudotime trajectory analysis
##monocle3
expression_matrix <- GetAssayData(ose2, assay = "RNA", layer = "counts")
cell_metadata <- ose2@meta.data
gene_annotation <- data.frame(gene_short_name = rownames(ose2))
rownames(gene_annotation) <- rownames(ose2)
cds <- new_cell_data_set(expression_matrix,
                         cell_metadata = cell_metadata,
                         gene_metadata = gene_annotation)
cds <- preprocess_cds(cds, num_dim = 50)
cds@int_colData$reducedDims$UMAP <- Embeddings(ose2, reduction = "harmony")
cds@clusters$UMAP$clusters <- ose2$seurat_clusters
names(cds@clusters$UMAP$clusters) <- colnames(cds)
recreate.partition <- c(rep(1, length(colnames(cds))))
names(recreate.partition) <- colnames(cds)
recreate.partition <- as.factor(recreate.partition)
cds@clusters$UMAP$partitions <- recreate.partition
cds <- learn_graph(cds, 
                   use_partition = TRUE,
                   close_loop = FALSE,
                   learn_graph_control = list(
                     ncenter = 140,                   
                     minimal_branch_len = 6,           
                     prune_graph = TRUE,                
                     orthogonal_proj_tip = FALSE,        
                     geodesic_distance_ratio = 1/3,      
                     euclidean_distance_ratio = 1        
                   ))
plot_cells(cds,color_cells_by = "stage",label_groups_by_cluster = FALSE,label_leaves = TRUE,label_branch_points = TRUE)
cds <- order_cells(cds, root_pr_nodes = get_earliest_principal_node(cds, celltype = "0"))
plot_cells(cds,
                 color_cells_by = "pseudotime",
                 label_cell_groups = FALSE,
                 label_leaves = FALSE,
                 label_branch_points = FALSE,
                 graph_label_size = 1.5)

##monocle2
MS_ct2 <- ose2@assays$RNA$counts
gene_ann <- data.frame(gene_short_name = row.names(MS_ct2), row.names = row.names(MS_ct2))
fd <- new("AnnotatedDataFrame",data=gene_ann)
pd <- new("AnnotatedDataFrame",data=ose2@meta.data)
MS_cds <- newCellDataSet(
  MS_ct2, 
  phenoData = pd,
  featureData =fd,
  expressionFamily = negbinomial.size(),
  lowerDetectionLimit=1)
  
MS_cds <- estimateSizeFactors(MS_cds)
MS_cds <- estimateDispersions(MS_cds)
MS_cds <- detectGenes(MS_cds, min_expr = 0.1) 
clustering_DEG_genes <- differentialGeneTest(MS_cds, fullModelFormulaStr = '~ose', cores = 4)
ordering_genes <- row.names(clustering_DEG_genes)[order(clustering_DEG_genes$qval)][1:50]
MS_cds <- setOrderingFilter(MS_cds, ordering_genes)
plot_ordering_genes(MS_cds)
MS_cds <-reduceDimension (MS_cds,max_components =2, method = 'DDRTree')
MS_cds <- orderCells(MS_cds)
gs <- c("Pou5f1","Trp63","Dlx5","Msx2","Pitx3","Prox1","Krt8","Pax6")
plot_cell_trajectory(MS_cds,markers = gs, use_color_gradient = T)
plot_cell_trajectory(MS_cds,color_by = "Pseudotime")
plot_cell_trajectory(MS_cds,color_by = "stage")
