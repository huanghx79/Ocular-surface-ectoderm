countsFile = c(Sample1 = "/Eye10P5",
               Sample2 = "/Eye8P5_9P5")
genecolumn = 2
prefix = "E8P5_10P5_MergedProject"
cellMinGene = 200
cellMaxGene = 10000
geneExpMinCell = 3
MaxMTPercent = 0.2
ScaleModel = "linear"
regressOut = "nCount_RNA,percent.mito"

pcadim = 1:10
computePCs = 50
nneighbors = 30
varGeneNum = 2000

Doubletrate = 0.1
pc.num_doublet = 1:10
pN = 0.25
doublet_resolution = 0.4

theta = 2
lambda = 1
resolution = 0.4

countsList = list()
for(sn in names(countsFile)){
	tmp = Read10X(data.dir = countsFile[sn], gene.column = genecolumn)
	colnames(tmp) = paste0(sn, "_", colnames(tmp))
	countsList[[sn]] = tmp
}

allcounts = countsList[[1]]
if(length(countsList) > 1){
	for(i in 2:length(countsList)){
		allcounts = RowMergeSparseMatrices(allcounts, countsList[[i]])
	}
}

cellSampleInfo = data.frame(
	cellBarcode = colnames(allcounts),
	sample = sapply(strsplit(colnames(allcounts), "_"), function(x) x[1])
)
rownames(cellSampleInfo) = cellSampleInfo$cellBarcode
allcounts = allcounts[,colSums(allcounts)>0,drop=F]
mito.genes = read.table("/MT.txt", sep = "\t", header = FALSE)[,1]
mito.genes = m```r
mito.genes = mito.genes[is.finite(match(mito.genes,rownames(allcounts)))]
percent.mito <- Matrix::colSums(allcounts[mito.genes, ])/Matrix::colSums(allcounts)
seuset <- CreateSeuratObject(counts = allcounts, project = prefix, min.cells = geneExpMinCell, min.features = 1)
seuset <- AddMetaData(seuset, metadata = cellSampleInfo[colnames(seuset),"sample"], col.name = "sample")
seuset <- subset(seuset, features = setdiff(row.names(seuset), mito.genes))
seuset <- AddMetaData(seuset, metadata = percent.mito, col.name = "percent.mito")
seuset <- subset(x = seuset, subset = nFeature_RNA > cellMinGene & nFeature_RNA < cellMaxGene)
seuset <- subset(x = seuset, subset = percent.mito < MaxMTPercent)

seuset_doublet <- seuset
seuset_doublet <- NormalizeData(seuset_doublet, normalization.method = "LogNormalize", scale.factor = 10000)
seuset_doublet <- FindVariableFeatures(seuset_doublet, selection.method = "vst", nfeatures = 2000)
seuset_doublet <- ScaleData(seuset_doublet, features = VariableFeatures(seuset_doublet))
seuset_doublet <- RunPCA(seuset_doublet, verbose = F)
seuset_doublet <- RunUMAP(seuset_doublet, dims = pc.num_doublet)
seuset_doublet <- FindNeighbors(seuset_doublet, dims = pc.num_doublet) %>% FindClusters(resolution = doublet_resolution)
sweep.res.list <- paramSweep_v3(seuset_doublet, PCs = pc.num_doublet)
sweep.stats <- summarizeSweep(sweep.res.list, GT = FALSE)
bcmvn <- find.pK(sweep.stats)
pK_bcmvn <- as.numeric(as.character(bcmvn$pK[which.max(bcmvn$BCmetric)]))
homotypic.prop <- modelHomotypic(seuset_doublet$seurat_clusters)
nExp_poi <- round(Doubletrate * ncol(seuset_doublet))
nExp_poi.adj <- round(nExp_poi * (1 - homotypic.prop))
seuset_doublet <- doubletFinder_v3(seuset_doublet, PCs = pc.num_doublet, pN = pN, pK = pK_bcmvn,
                                     nExp = nExp_poi.adj, reuse.pANN = F)
df_col <- tail(grep("^DF.classifications", colnames(seuset_doublet@meta.data), value = TRUE), 1)
doublet_cells <- colnames(seuset_doublet)[seuset_doublet@meta.data[[df_col]] == "Doublet"]
seuset <- subset(seuset, cells = setdiff(colnames(seuset), doublet_cells))

seuset <- NormalizeData(seuset, normalization.method = "LogNormalize", scale.factor = 10000)
seuset <- FindVariableFeatures(seuset, selection.method = "vst", nfeatures = varGeneNum)
scalegene = VariableFeatures(seuset)
seuset <- ScaleData(object = seuset, features = scalegene, vars.to.regress = regressOut, model.use = ScaleModel)
seuset <- RunPCA(object = seuset, features = VariableFeatures(seuset), npcs = computePCs, do.print = TRUE, ndims.print = 1:computePCs, nfeatures.print = 10)
seuset = RunHarmony(seuset, "sample", assay.use = DefaultAssay(seuset), theta = theta, lambda = lambda)
seuset <- RunUMAP(seuset, dims = pcadim, n.neighbors = nneighbors, reduction = "harmony")
seuset <- FindNeighbors(seuset, reduction = "harmony", dims = pcadim)
seuset <- FindClusters(seuset, resolution = resolution)
E8P5_10P5_SE <- subset(x = seuset, idents = "7")
saveRDS(E8P5_10P5_SE, file="./E8P5_10P5_SE.Harmony.rds"
